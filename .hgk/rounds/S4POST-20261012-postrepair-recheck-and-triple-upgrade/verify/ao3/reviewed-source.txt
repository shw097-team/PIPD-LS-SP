def _png_chunk(chunk_type: bytes, data: bytes) -> bytes:
    crc = zlib.crc32(chunk_type + data) & 0xFFFFFFFF
    return struct.pack(">I", len(data)) + chunk_type + data + struct.pack(">I", crc)


def png_icon_bytes(size: int = 128) -> bytes:
    """A deterministic RGBA icon: no Pillow, no new dependency, no wall-clock input."""
    if size < 8:
        raise ValueError("icon size must be at least 8 px")
    raw = bytearray()
    for y in range(size):
        raw.append(0)  # filter type 0 (None)
        for x in range(size):
            r = (x * 255) // (size - 1)
            g = (y * 255) // (size - 1)
            border = x < size // 16 or x >= size - size // 16 or y < size // 16 or y >= size - size // 16
            b = 0x10 if border else 0x60
            raw += bytes((r, g, b, 255))
    ihdr = struct.pack(">IIBBBBB", size, size, 8, 6, 0, 0, 0)
    return (PNG_SIG + _png_chunk(b"IHDR", ihdr)
            + _png_chunk(b"IDAT", zlib.compress(bytes(raw), 9))
            + _png_chunk(b"IEND", b""))


def png_errors(data: bytes) -> list[str]:
    """Decode the PNG structurally: signature, chunk CRCs, IHDR and an inflated IDAT."""
    if len(data) < 8:
        return ["too_short"]
    if data[:8] != PNG_SIG:
        return ["signature"]
    errors: list[str] = []
    pos = 8
    width = height = bit_depth = color_type = None
    idat = bytearray()
    seen_ihdr = seen_iend = False
    first_chunk = None
    while pos < len(data):
        if pos + 8 > len(data):
            errors.append("truncated_chunk_header")
            break
        length = struct.unpack(">I", data[pos:pos + 4])[0]
        chunk_type = data[pos + 4:pos + 8]
        pos += 8
        if pos + length + 4 > len(data):
            errors.append(f"truncated_chunk_body:{chunk_type.decode('latin1')}")
            break
        body = data[pos:pos + length]
        crc = data[pos + length:pos + length + 4]
        pos += length + 4
        if first_chunk is None:
            first_chunk = chunk_type
            if chunk_type != b"IHDR":
                errors.append("ihdr_not_first")
        if crc != struct.pack(">I", zlib.crc32(chunk_type + body) & 0xFFFFFFFF):
            errors.append(f"crc:{chunk_type.decode('latin1')}")
        if chunk_type == b"IHDR":
            seen_ihdr = True
            if len(body) != 13:
                errors.append("ihdr_length")
            else:
                width, height, bit_depth, color_type, _comp, _filt, interlace = struct.unpack(
                    ">IIBBBBB", body)
                if width == 0 or height == 0:
                    errors.append("ihdr_zero_dimension")
                if bit_depth != 8:
                    errors.append("ihdr_bit_depth")
                if color_type not in (0, 2, 4, 6):
                    errors.append("ihdr_color_type")
                if interlace != 0:
                    errors.append("ihdr_interlace")
        elif chunk_type == b"IDAT":
            idat += body
        elif chunk_type == b"IEND":
            seen_iend = True
            break
    if not seen_ihdr:
        errors.append("ihdr_missing")
    if not seen_iend:
        errors.append("iend_missing")
    if seen_ihdr and seen_iend and width and height:
        channels = {0: 1, 2: 3, 4: 2, 6: 4}.get(color_type, 0)
        try:
            inflated = zlib.decompress(bytes(idat))
        except zlib.error:
            errors.append("idat_inflate")
        else:
            if channels == 0 or len(inflated) != height * (1 + width * channels):
                errors.append("idat_size")
    return errors


# --------------------------------------------------------------------------- JSON-schema subset
def _type_ok(value, typ: str) -> bool:
    if typ == "object":
        return isinstance(value, dict)
    if typ == "array":
        return isinstance(value, list)
    if typ == "string":
        return isinstance(value, str)
    if typ == "integer":
        return isinstance(value, int) and not isinstance(value, bool)
    if typ == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if typ == "boolean":
        return isinstance(value, bool)
    if typ == "null":
        return value is None
    return True


def validate_subset(instance, schema, path: str = "$") -> list[dict]:
    """A tiny, dependency-free JSON-Schema subset validator (local subset AND vendor schema).

    Supported keywords: ``type``, ``const``, ``enum``, ``minLength``, ``maxLength``, ``pattern``,
    ``minimum``, ``maximum``, ``minItems``, ``maxItems``, ``uniqueItems``, ``items``, ``required``,
    ``properties`` and both forms of ``additionalProperties`` (``false`` closes the key set; an
    object schema validates every key that ``properties`` does not name).
    """
    errors: list[dict] = []
    if not isinstance(schema, dict):
        return errors
    if "type" in schema:
        types = schema["type"] if isinstance(schema["type"], list) else [schema["type"]]
        if not any(_type_ok(instance, t) for t in types):
            return [{"path": path, "keyword": "type", "message": f"expected one of {types}"}]
    if "const" in schema and instance != schema["const"]:
        errors.append({"path": path, "keyword": "const",
                       "message": f"expected const {schema['const']!r}"})
    if "enum" in schema and instance not in schema["enum"]:
        errors.append({"path": path, "keyword": "enum",
                       "message": f"expected one of {schema['enum']!r}"})
    if isinstance(instance, str):
        if "minLength" in schema and len(instance) < schema["minLength"]:
            errors.append({"path": path, "keyword": "minLength", "message": "too short"})
        if "maxLength" in schema and len(instance) > schema["maxLength"]:
            errors.append({"path": path, "keyword": "maxLength", "message": "too long"})
        if "pattern" in schema and re.search(schema["pattern"], instance) is None:
            errors.append({"path": path, "keyword": "pattern",
                           "message": f"does not match {schema['pattern']!r}"})
    if isinstance(instance, int) and not isinstance(instance, bool):
        if "minimum" in schema and instance < schema["minimum"]:
            errors.append({"path": path, "keyword": "minimum", "message": "below minimum"})
        if "maximum" in schema and instance > schema["maximum"]:
            errors.append({"path": path, "keyword": "maximum", "message": "above maximum"})
    if isinstance(instance, list):
        if "minItems" in schema and len(instance) < schema["minItems"]:
            errors.append({"path": path, "keyword": "minItems", "message": "too few items"})
        if "maxItems" in schema and len(instance) > schema["maxItems"]:
            errors.append({"path": path, "keyword": "maxItems", "message": "too many items"})
        if schema.get("uniqueItems"):
            seen = {json.dumps(item, sort_keys=True) for item in instance}
            if len(seen) != len(instance):
                errors.append({"path": path, "keyword": "uniqueItems", "message": "duplicates"})
        items = schema.get("items")
        if isinstance(items, dict):
            for i, item in enumerate(instance):
                errors += validate_subset(item, items, f"{path}[{i}]")
    if isinstance(instance, dict):
        for required_key in schema.get("required", []):
            if required_key not in instance:
                errors.append({"path": f"{path}.{required_key}", "keyword": "required",
                               "message": "required property is absent"})
        props = schema.get("properties", {})
        extra = schema.get("additionalProperties")
        if extra is False:
            for key in instance:
                if key not in props:
                    errors.append({"path": f"{path}.{key}", "keyword": "additionalProperties",
                                   "message": "additional property is not allowed"})
        for key, value in instance.items():
            if key in props:
                errors += validate_subset(value, props[key], f"{path}.{key}")
            elif isinstance(extra, dict):
                errors += validate_subset(value, extra, f"{path}.{key}")
    return errors


# --------------------------------------------------------------------------- frontmatter
def _unquote(value: str) -> str:
    value = value.strip()
    if len(value) >= 2 and value[0] == value[-1] and value[0] in ("'", '"'):
        return value[1:-1]
    return value


def parse_skill_md(text: str) -> tuple[list[str], dict, str]:
    """Split ``SKILL.md`` into (ordered keys, mapping, body). Never mutates the body text."""
    first_nl = text.find("\n")
    if first_nl == -1 or text[:first_nl].strip() != "---":
        raise Refusal("SKILL_FRONTMATTER_INVALID", "SKILL.md does not open with a `---` line")
    pos = first_nl + 1
    fm_lines: list[str] = []
    body: str | None = None
    while True:
        nl = text.find("\n", pos)
        if nl == -1:
            if text[pos:].strip() == "---":
                body = ""
                break
            raise Refusal("SKILL_FRONTMATTER_INVALID", "frontmatter is not terminated by `---`")
        line = text[pos:nl]
        if line.strip() == "---":
            body = text[nl + 1:]
            break
        fm_lines.append(line)
        pos = nl + 1
    mapping: dict[str, object] = {}
    keys: list[str] = []
    current: str | None = None
    for raw in fm_lines:
        line = raw.rstrip("\r")
        if not line.strip() or line.lstrip().startswith("#"):
            continue
        if line.lstrip().startswith("- "):
            if current is None:
                raise Refusal("SKILL_FRONTMATTER_INVALID", "list item without a preceding key")
            cast = mapping[current]
            if not isinstance(cast, list):
                raise Refusal("SKILL_FRONTMATTER_INVALID", f"key {current!r} mixes scalar and list")
            cast.append(_unquote(line.lstrip()[2:]))
            continue
        if ":" not in line:
            raise Refusal("SKILL_FRONTMATTER_INVALID",
                          f"frontmatter line is not `key: value`: {line!r}")
        key, _, value = line.partition(":")
        key = key.strip()
        value = value.strip()
        if not key:
            raise Refusal("SKILL_FRONTMATTER_INVALID", f"empty frontmatter key: {line!r}")
        if key in mapping:
            raise Refusal("SKILL_FRONTMATTER_INVALID", f"duplicate frontmatter key {key!r}")
        if value == "":
            mapping[key] = []
            current = key
        else:
            mapping[key] = _unquote(value)
            current = None
        keys.append(key)
    if not keys:
        raise Refusal("SKILL_FRONTMATTER_INVALID", "frontmatter carries no keys")
    if body is None:
        raise Refusal("SKILL_FRONTMATTER_INVALID", "frontmatter has no body")
    return keys, mapping, body


def _emit_frontmatter(mapping: dict, keys: list[str]) -> str:
    lines = ["---"]
    for key in keys:
        value = mapping[key]
        if isinstance(value, list):
            lines.append(f"{key}:")
            lines.extend(f"  - {item}" for item in value)
        else:
            lines.append(f"{key}: {value}")
    lines.append("---")
    return "\n".join(lines) + "\n"


def project_skill_md(skill: str, source_bytes: bytes, record: dict) -> tuple[bytes, dict]:
    """Project one derived ``SKILL.md``: keep portable keys only, body byte-identical."""
    try:
        text = source_bytes.decode("utf-8")
    except UnicodeDecodeError as exc:
        raise Refusal("SKILL_NOT_UTF8", f"{skill}/SKILL.md is not UTF-8: {exc}") from exc
    keys, mapping, body = parse_skill_md(text)
    # The frontmatter `name` is the skill's portable identity: it must name this skill directory.
    if mapping.get("name") != skill:
        raise Refusal("SKILL_FRONTMATTER_INVALID",
                      f"{skill}: frontmatter `name` is {mapping.get('name')!r}; "
                      f"it must be the skill name {skill!r}")
    kept = [k for k in keys if k in PORTABLE_KEYS]
    dropped = [k for k in keys if k not in PORTABLE_KEYS]
    if set(record.get("kept_keys", [])) != set(kept):
        raise Refusal("DERIVATION_KEY_SET_MISMATCH",
                      f"{skill}: record kept_keys {record.get('kept_keys')} != observed {kept}")
    if set(record.get("dropped_keys", [])) != set(dropped):
        raise Refusal("DERIVATION_KEY_SET_MISMATCH",
                      f"{skill}: record dropped_keys {record.get('dropped_keys')} != observed {dropped}")
    projected = (_emit_frontmatter(mapping, kept) + body).encode("utf-8")
    # The body must survive the projection byte-for-byte.
    _pk, _pm, projected_body = parse_skill_md(projected.decode("utf-8"))
    if projected_body.encode("utf-8") != body.encode("utf-8"):
        raise Refusal("PROJECTION_BODY_CHANGED", f"{skill}: frontmatter projection changed the body")
    return projected, {"kept_keys": kept, "dropped_keys": dropped}


# --------------------------------------------------------------------------- source walking
def _list_skill_dirs(skills_root: Path) -> dict[str, Path]:
    if not skills_root.is_dir():
        raise Refusal("SKILLS_ROOT_MISSING", str(skills_root))
    out: dict[str, Path] = {}
    for child in sorted(skills_root.iterdir()):
        if child.name.startswith(".") or child.name == "__pycache__":
            continue
        if _path_is_symlink(child):
            raise Refusal("SOURCE_SYMLINK", str(child))
        if not child.is_dir():
            raise Refusal("SOURCE_SPECIAL_MEMBER", f"{child} is not a skill directory")
        out[child.name] = child
    return out


def _walk_source_files(repo_root: Path, skill_dir: Path) -> dict[str, bytes]:
    files: dict[str, bytes] = {}
    for dirpath, dirnames, filenames in os.walk(skill_dir, topdown=True, followlinks=False):
        dirnames.sort()
        filenames.sort()
        for d in list(dirnames):
            p = Path(dirpath) / d
            if _path_is_symlink(p):
                raise Refusal("SOURCE_SYMLINK", str(p))
            if d.startswith(".") or d == "__pycache__":
                raise Refusal("HIDDEN_OR_SPECIAL_MEMBER", str(p))
        for f in filenames:
            p = Path(dirpath) / f
            if _path_is_symlink(p):
                raise Refusal("SOURCE_SYMLINK", str(p))
            if not p.is_file():
                raise Refusal("SOURCE_SPECIAL_MEMBER", str(p))
            if f.startswith("."):
                raise Refusal("HIDDEN_OR_SPECIAL_MEMBER", str(p))
            files[p.relative_to(repo_root).as_posix()] = canonical_bytes(p.read_bytes())
    return files


# --------------------------------------------------------------------------- output-root gates
def _find_symlink_ancestor(path: Path) -> Path | None:
    for anc in (path, *path.parents):
        if _path_is_symlink(anc):
            return anc
    return None


def check_output_root(out_dir: Path, source_root: Path) -> None:
    symlink = _find_symlink_ancestor(out_dir)
    if symlink is not None:
        raise Refusal("OUTPUT_ROOT_SYMLINK", f"{symlink} is a symlink; refusing to write through it")
    out_res = out_dir.resolve()
    repo_res = source_root.resolve()
    dist_res = (repo_res / "dist").resolve()
    if out_res == dist_res or dist_res in out_res.parents:
        raise Refusal("OUTPUT_ROOT_IN_DIST",
                      f"output root {out_res} is inside the release directory {dist_res}")
    if out_res == repo_res or repo_res in out_res.parents:
        raise Refusal("OUTPUT_ROOT_OVERLAPS_REPO",
                      f"output root {out_res} overlaps the repository {repo_res}")
    if out_res in repo_res.parents:
        raise Refusal("OUTPUT_ROOT_OVERLAPS_REPO",
                      f"output root {out_res} is an ancestor of the repository {repo_res}")
    if os.name == "posix" and out_dir.exists() and (os.stat(out_dir).st_mode & 0o002):
        raise Refusal("OUTPUT_ROOT_WORLD_WRITABLE", f"output root {out_dir} is world-writable")


# --------------------------------------------------------------------------- ZIP
def _write_zip(path: Path, members: dict[str, bytes]) -> None:
    with zipfile.ZipFile(path, "w") as zf:
        for arcname in sorted(members):
            info = zipfile.ZipInfo(arcname, date_time=FIXED_DATE_TIME)
            info.compress_type = zipfile.ZIP_DEFLATED
            info.external_attr = FIXED_MODE << 16
            info.create_system = 0
            zf.writestr(info, members[arcname], compress_type=zipfile.ZIP_DEFLATED,
                        compresslevel=COMPRESS_LEVEL)


def _exclusive_write(path: Path, data: bytes) -> None:
    """Create a release sidecar exclusively; refuse a pre-existing path of any kind.

    ``Path.write_bytes`` opens an existing name and truncates the inode behind it, so an attacker who
    can prepopulate the output directory - with a symlink, or with a hardlink to a file outside it -
    can make the builder overwrite that file. Measured against this tool as AO-01 of the round's
    independent acceptance challenge. ``O_EXCL`` refuses any pre-existing name (a hardlink is an
    ordinary regular file, so only exclusivity closes it) and ``O_NOFOLLOW`` refuses a symlink.
    """
    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL | getattr(os, "O_NOFOLLOW", 0)
    try:
        fd = os.open(path, flags, 0o644)
    except FileExistsError as exc:
        raise Refusal("OUTPUT_MEMBER_EXISTS",
                      f"{path} already exists; refusing to overwrite an existing release member") from exc
    except OSError as exc:
        raise Refusal("OUTPUT_MEMBER_UNSAFE",
                      f"{path} cannot be created exclusively: {exc}") from exc
    try:
        with os.fdopen(fd, "wb") as fh:
            fh.write(data)
            fh.flush()
            os.fsync(fh.fileno())
    except BaseException:
        try:
            path.unlink()
        except OSError:
            pass
        raise


def _verify_zip(path: Path, members: dict[str, bytes]) -> None:
    with zipfile.ZipFile(path) as zf:
        infos = zf.infolist()
        if [i.filename for i in infos] != sorted(members):
            raise Refusal("INTERNAL_ZIP_MISMATCH", "member names or order differ from the plan")
        for info in infos:
            if info.date_time != FIXED_DATE_TIME:
                raise Refusal("INTERNAL_ZIP_MISMATCH", f"non-fixed date_time on {info.filename}")
            if (info.external_attr >> 16) != FIXED_MODE:
                raise Refusal("INTERNAL_ZIP_MISMATCH", f"non-fixed mode on {info.filename}")
            if info.compress_type != zipfile.ZIP_DEFLATED:
                raise Refusal("INTERNAL_ZIP_MISMATCH", f"unexpected compression on {info.filename}")
            if zf.read(info.filename) != members[info.filename]:
                raise Refusal("INTERNAL_ZIP_MISMATCH", f"member bytes differ for {info.filename}")

def _load_json(path: Path, code: str) -> dict:
    if not path.is_file():
        raise Refusal(code, f"required input is absent: {path}")
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (OSError, ValueError) as exc:
        raise Refusal(code, f"{path} is not parseable JSON: {exc}") from exc


def _load_vendor_schema(packaging_root: Path) -> tuple[bytes, dict]:
    """Load the byte-locked official vendor schema: the lock is the ``$id`` AND the exact bytes.

    The bytes fetched once from the Agent Plugins 1.0 specification are committed next to the
    manifests, so this gate can be re-run offline; it is NOT the vendor's own validator, which
    stays NOT_EXECUTED.
    """
    path = packaging_root / VENDOR_SCHEMA_FILENAME
    if not path.is_file():
        raise Refusal("VENDOR_SCHEMA_MISSING", f"the locked vendor schema is absent: {path}")
    data = path.read_bytes()
    try:
        doc = json.loads(data.decode("utf-8"))
    except (UnicodeDecodeError, ValueError) as exc:
        raise Refusal("VENDOR_SCHEMA_INVALID", f"{path} is not parseable JSON: {exc}") from exc
    if not isinstance(doc, dict):
        raise Refusal("VENDOR_SCHEMA_INVALID", f"{path} is not a JSON object")
    if doc.get("$id") != VENDOR_SCHEMA_ID:
        raise Refusal("VENDOR_SCHEMA_ID_MISMATCH",
                      f"vendor schema $id is {doc.get('$id')!r}; expected {VENDOR_SCHEMA_ID!r}")
    observed = sha256_bytes(data)
    if observed != VENDOR_SCHEMA_SHA256:
        raise Refusal("VENDOR_SCHEMA_DIGEST_MISMATCH",
                      f"vendor schema bytes hash to {observed}; the locked digest is "
                      f"{VENDOR_SCHEMA_SHA256}")
    return data, doc


def _check_owner_inputs(release: dict, vendor_schema_bytes: bytes, derivations_doc: dict,
                        skill_paths: dict[str, str], ledger_files: dict[str, str]) -> dict[str, dict]:
    """Gate every owner input. The FIRST missing input raises, in canonical skill order."""
    records = derivations_doc.get("derivations")
    if not isinstance(records, list):
        raise Refusal("DERIVATIONS_INVALID", "derivations.json must carry a `derivations` list")
    by_skill: dict[str, dict] = {}
    for record in records:
        if isinstance(record, dict) and isinstance(record.get("skill"), str):
            by_skill[record["skill"]] = record

    for skill in sorted(EXPECTED_SKILLS):
        record = by_skill.get(skill)
        if record is None:
            raise Refusal("OWNER_INPUT_MISSING", DERIVATION_MISSING_TEMPLATE.format(skill=skill))
        source_rel = skill_paths[skill]
        if record.get("source_path") != source_rel:
            raise Refusal("DERIVATION_RECORD_INVALID",
                          f"{skill}: source_path {record.get('source_path')!r} != {source_rel!r}")
        if record.get("source_sha256") != ledger_files.get(source_rel):
            raise Refusal("DERIVATION_SOURCE_MISMATCH",
                          f"{skill}: source_sha256 does not match the frozen ledger")
        if is_placeholder(record.get("approved_by")) or is_placeholder(record.get("approval_receipt")):
            raise Refusal("DERIVATION_RECORD_INVALID",
                          f"{skill}: approved_by / approval_receipt must name the owner approval")
        if not isinstance(record.get("kept_keys"), list) or not isinstance(record.get("dropped_keys"), list):
            raise Refusal("DERIVATION_RECORD_INVALID", f"{skill}: kept_keys/dropped_keys must be lists")

    pinned = release.get("official_schema_sha256")
    if is_placeholder(pinned):
        raise Refusal("OWNER_INPUT_MISSING",
                      "official_schema_sha256 (the digest of the official Agent Plugins vendor "
                      "schema) is not pinned (value is a placeholder)")
    if not HEX64.match(str(pinned)):
        raise Refusal("OFFICIAL_SCHEMA_DIGEST_INVALID",
                      f"official_schema_sha256 is not 64 hex chars: {pinned!r}")
    observed = sha256_bytes(vendor_schema_bytes)
    if pinned != observed:
        raise Refusal("VENDOR_SCHEMA_DIGEST_MISMATCH",
                      f"pinned official_schema_sha256 {pinned} does not match the vendor schema "
                      f"bytes it validates against ({observed})")

    publisher = release.get("publisher") or {}
    if any(is_placeholder(publisher.get(key)) for key in ("name", "identity", "identity_sha256")):
        raise Refusal("OWNER_INPUT_MISSING",
                      "publisher identity is not pinned (name/identity/identity_sha256 are placeholders)")
    if not HEX64.match(str(publisher.get("identity_sha256"))):
        raise Refusal("PUBLISHER_IDENTITY_INVALID", "publisher.identity_sha256 is not 64 hex chars")

    icons = release.get("icons")
    if not isinstance(icons, list) or not icons:
        raise Refusal("OWNER_INPUT_MISSING", "the release bookkeeping pins no icons")
    for icon in icons:
        rel = icon.get("path")
        if is_placeholder(icon.get("sha256")):
            raise Refusal("OWNER_INPUT_MISSING", f"icon digest for '{rel}' is not pinned")
    return by_skill


def build_plugin(*, source_root: Path, ledger: dict, source_commit: str | None,
                 packaging_root: Path, out_dir: Path, generate_icons: bool = False,
                 expected_skills: tuple[str, ...] = EXPECTED_SKILLS) -> dict:
    source_root = Path(source_root).resolve()
    packaging_root = Path(packaging_root).resolve()
    out_dir = Path(out_dir)

    # --- identity gates -----------------------------------------------------------------
    if source_commit is None:
        raise Refusal("SOURCE_COMMIT_NULL", "the pinned source commit is null; refusing to build")
    if not HEX40.match(str(source_commit)):
        raise Refusal("SOURCE_COMMIT_INVALID", f"source commit is not 40 hex chars: {source_commit!r}")
    if source_commit != ledger.get("commit"):
        raise Refusal("SOURCE_COMMIT_MISMATCH",
                      f"source commit {source_commit} != ledger commit {ledger.get('commit')}")

    # --- output-root gates --------------------------------------------------------------
    check_output_root(out_dir, source_root)

    # --- source set vs the frozen ledger ------------------------------------------------
    skills_root = source_root / "skills"
    dirs = _list_skill_dirs(skills_root)
    unexpected = sorted(n for n in dirs if n not in set(expected_skills))
    if unexpected:
        raise Refusal("UNEXPECTED_SKILL",
                      f"the plugin is exactly 8 skills; found extra skill(s) {unexpected}")
    missing_skill = sorted(n for n in expected_skills if n not in dirs)
    if missing_skill:
        raise Refusal("MISSING_SKILL", f"frozen ledger expects skill(s) {missing_skill}")
    if sorted(dirs) != sorted(ledger.get("skills", [])):
        raise Refusal("SKILL_SET_CHANGED", "on-disk skill set != frozen ledger skill set")

    disk_files: dict[str, bytes] = {}
    for name in sorted(dirs):
        disk_files.update(_walk_source_files(source_root, dirs[name]))
    ledger_files = dict(ledger.get("files", {}))
    extra = sorted(set(disk_files) - set(ledger_files))
    missing = sorted(set(ledger_files) - set(disk_files))
    if extra or missing:
        raise Refusal("SOURCE_SET_CHANGED",
                      f"source set changed vs the frozen ledger; extra={extra} missing={missing}")
    for rel in sorted(ledger_files):
        if sha256_bytes(disk_files[rel]) != ledger_files[rel]:
            raise Refusal("SOURCE_DIGEST_MISMATCH", f"{rel} does not match the frozen ledger")

    # --- the official root manifest + the locked vendor schema ---------------------------
    manifest = _load_json(packaging_root / "plugin.json", "MANIFEST_MISSING")
    release = _load_json(packaging_root / RELEASE_BOOKKEEPING_FILENAME,
                         "RELEASE_BOOKKEEPING_MISSING")
    vendor_bytes, vendor_doc = _load_vendor_schema(packaging_root)
    # (a) the official manifest must name the vendor schema it is validated against
    if manifest.get("$schema") != VENDOR_SCHEMA_ID:
        raise Refusal("OFFICIAL_SCHEMA_ID_MISMATCH",
                      f"the root manifest `$schema` must be the vendor $id {VENDOR_SCHEMA_ID!r}, "
                      f"got {manifest.get('$schema')!r}")
    # (b) validate the official manifest against the vendor schema with our own subset validator
    violations = validate_subset(manifest, vendor_doc)
    if violations:
        first = violations[0]
        raise Refusal("MANIFEST_SCHEMA_VIOLATION",
                      f"{first['path']}: {first['keyword']} — {first['message']}")

    # --- PIPD's release bookkeeping + the local subset ------------------------------------
    schema_path = packaging_root / "plugin.schema.json"
    schema_bytes = schema_path.read_bytes() if schema_path.is_file() else None
    if schema_bytes is None:
        raise Refusal("SCHEMA_MISSING", f"local schema subset is absent: {schema_path}")
    try:
        schema_doc = json.loads(schema_bytes.decode("utf-8"))
    except (UnicodeDecodeError, ValueError) as exc:
        raise Refusal("SCHEMA_INVALID", f"{schema_path} is not parseable JSON: {exc}") from exc
    violations = validate_subset(release, schema_doc)
    if violations:
        first = violations[0]
        raise Refusal("MANIFEST_SCHEMA_VIOLATION",
                      f"plugin.release.json {first['path']}: {first['keyword']} — {first['message']}")
    if sorted(release.get("skills", [])) != sorted(expected_skills):
        raise Refusal("MANIFEST_SCHEMA_VIOLATION",
                      "the release bookkeeping skills must name exactly the eight frozen skills")
    embedded = (manifest.get("extensions") or {}).get(RELEASE_EXTENSION_KEY)
    if embedded != release:
        raise Refusal("RELEASE_EXTENSION_MISMATCH",
                      f"the root manifest extensions[{RELEASE_EXTENSION_KEY!r}] must be the "
                      f"{RELEASE_BOOKKEEPING_FILENAME} object verbatim")

    derivations_doc = _load_json(packaging_root / "derivations.json", "DERIVATIONS_MISSING")

    # --- icons (owner-gated, validated after generation) --------------------------------
    icons = release.get("icons") or []
    if generate_icons:
        for icon in icons:
            target = packaging_root / icon["path"]
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(png_icon_bytes(int(icon.get("size", 128))))

    # --- owner-input gate (FIRST missing input refuses, deterministically) --------------
    skill_paths = {name: f"skills/{name}/SKILL.md" for name in expected_skills}
    records = _check_owner_inputs(release, vendor_bytes, derivations_doc, skill_paths, ledger_files)

    for icon in icons:
        target = packaging_root / icon["path"]
        if not target.is_file() or _path_is_symlink(target):
            raise Refusal("ICON_MISSING", f"pinned icon is absent or not a regular file: {target}")
        data = target.read_bytes()
        if sha256_bytes(data) != icon["sha256"]:
            raise Refusal("ICON_DIGEST_MISMATCH", f"{icon['path']} does not match its pinned digest")
        problems = png_errors(data)
        if problems:
            raise Refusal("ICON_INVALID", f"{icon['path']}: {problems[0]}")

    # --- assemble the plugin member set -------------------------------------------------
    plugin_id = str(release["plugin_id"])
    root = plugin_id
    members: dict[str, bytes] = {}
    mapping_rows: list[dict] = []
    for name in sorted(expected_skills):
        skill_prefix = f"skills/{name}/"
        for rel in sorted(r for r in disk_files if r.startswith(skill_prefix)):
            rest = rel[len(skill_prefix):]
            dest = f"{root}/{rel}"
            if rest == "SKILL.md":
                dest_bytes, _ = project_skill_md(name, disk_files[rel], records[name])
                mode = "DERIVED_WITH_OWNER_APPROVAL"
            else:
                dest_bytes = disk_files[rel]
                mode = "EXACT"
            members[dest] = dest_bytes
            mapping_rows.append({
                "source_path": rel,
                "source_sha256": ledger_files[rel],
                "destination_path": dest,
                "destination_sha256": sha256_bytes(dest_bytes),
                "mode": mode,
            })
    members[f"{root}/plugin.json"] = canonical_json_bytes(manifest)
    members[f"{root}/{RELEASE_BOOKKEEPING_FILENAME}"] = canonical_json_bytes(release)
    members[f"{root}/plugin.schema.json"] = schema_bytes
    members[f"{root}/{VENDOR_SCHEMA_FILENAME}"] = vendor_bytes
    members[f"{root}/derivations.json"] = canonical_json_bytes(derivations_doc)
    for icon in icons:
        members[f"{root}/{icon['path']}"] = (packaging_root / icon["path"]).read_bytes()

    for arcname in members:
        for part in arcname.split("/"):
            if part.startswith(".") or part in ("__pycache__", "Thumbs.db"):
                raise Refusal("HIDDEN_OR_SPECIAL_MEMBER", f"refusing to package {arcname!r}")

    mapping_doc = {
        "schema": "PIPD-PUBLIC-PLUGIN-SOURCE-MAPPING/1",
        "plugin_id": plugin_id,
        "source_commit": source_commit,
        "source_tree": ledger.get("tree"),
        "source_ledger_version": ledger.get("ledger_version"),
        "entries": mapping_rows,
    }
    mapping_bytes = canonical_json_bytes(mapping_doc)
    members[f"{root}/SOURCE_TO_PLUGIN_MAPPING.json"] = mapping_bytes
    receipt_doc = {
        "schema": "PIPD-PUBLIC-PLUGIN-RELEASE-RECEIPT/1",
        "plugin_id": plugin_id,
        "version": release.get("version"),
        "source_commit": source_commit,
        "source_tree": ledger.get("tree"),
        "source_ledger_version": ledger.get("ledger_version"),
        "official_schema_sha256": str(release.get("official_schema_sha256")),
        "official_schema_id": VENDOR_SCHEMA_ID,
        "claim_ceiling": "LOCAL_PACKAGER_PASS_ONLY",
        "official_portal_verification": "NOT_EXECUTED",
        "member_count": len(members) + 1,
        "mapping_sha256": sha256_bytes(mapping_bytes),
        "determinism": {"zip_mtime": "1980-01-01T00:00:00Z", "mode": "0644",
                        "compression": "DEFLATED:9"},
    }
    receipt_bytes = canonical_json_bytes(receipt_doc)
    members[f"{root}/RELEASE_RECEIPT.json"] = receipt_bytes

    # --- deterministic ZIP, read-back verify, then promote ------------------------------
    out_dir.mkdir(parents=True, exist_ok=True)
    if _path_is_symlink(out_dir):
        raise Refusal("OUTPUT_ROOT_SYMLINK", f"{out_dir} became a symlink; refusing to write")
    final_zip = out_dir / f"{plugin_id}.zip"
    if _path_is_symlink(final_zip) or final_zip.exists():
        raise Refusal("OUTPUT_ARTIFACT_EXISTS",
                      f"{final_zip} already exists; refusing to replace an existing release artifact")
    fd, tmp_name = tempfile.mkstemp(prefix=".public-plugin-", suffix=".zip", dir=str(out_dir))
    os.close(fd)
    tmp_zip = Path(tmp_name)
    sidecar = final_zip.with_name(final_zip.name + ".sha256")
    mapping_path = out_dir / "SOURCE_TO_PLUGIN_MAPPING.json"
    receipt_path = out_dir / "RELEASE_RECEIPT.json"
    promoted = False
    created: list[Path] = []
    try:
        _write_zip(tmp_zip, members)
        _verify_zip(tmp_zip, members)
        # The sidecars are created EXCLUSIVELY and BEFORE the artifact is promoted, so a pre-seeded
        # output member refuses the whole build while nothing has been published yet. Only members
        # this call actually created are rolled back: a member that was already there (and caused the
        # refusal) is left exactly as the attacker/operator left it, never deleted.
        digest = sha256_bytes(tmp_zip.read_bytes())
        for target, payload in ((sidecar, f"{digest}  {final_zip.name}\n".encode("utf-8")),
                                (mapping_path, mapping_bytes),
                                (receipt_path, receipt_bytes)):
            _exclusive_write(target, payload)
            created.append(target)
        os.replace(tmp_zip, final_zip)
        promoted = True
    finally:
        if tmp_zip.exists():
            tmp_zip.unlink()
        if not promoted:
            for partial in created:
                if partial.exists() and not _path_is_symlink(partial):
                    try:
                        partial.unlink()
                    except OSError:
                        pass

    return {
        "verdict": "BUILT",
        "out_zip": str(final_zip),
        "zip_sha256": digest,
        "sidecar": str(sidecar),
        "member_count": len(members),
        "mapping": str(mapping_path),
        "receipt": str(receipt_path),
        "claim_ceiling": "LOCAL_PACKAGER_PASS_ONLY",
        "official_portal_verification": "NOT_EXECUTED",
    }

#!/usr/bin/env python3
"""S4POST Stage B: project one frozen Web mount pack into ONE bounded fallback document.

There is no public directory listing for the plugin yet, so the current Web route is a single
markdown document a reader can attach to a conversation or a Knowledge base. This tool builds that
document from a *frozen mount pack* (a directory carrying ``MANIFEST.sha256``) and is fail-closed:
it REFUSES (typed reason, non-zero exit) rather than warn on

  * ``MANIFEST_DIGEST_MISMATCH``  a member's bytes no longer match ``MANIFEST.sha256``
  * ``MEMBER_SYMLINK``            a symlinked member (or pack) is present
  * ``PATH_ESCAPE``               a manifest path escapes the pack root
  * ``UNEXPECTED_EXTRA_FILE``     a file exists in the pack that the manifest does not list
  * ``RELEASE_STAMP_MISSING``     a member document does not stamp the pinned release id AND the
                                  pinned commit (a member naming only one of the two is stale)
  * ``MEMBER_NOT_TEXT``           a member is not UTF-8 text (it could not be embedded verbatim)
  * ``SIZE_CAP_EXCEEDED``         the composed document would exceed the cap (NEVER truncated)
  * ``DESTINATION_EXISTS`` / ``DESTINATION_SYMLINK``  the output file already exists

The document is written with ``O_EXCL`` and stamps ``DOC_MOUNT_ONLY`` plus the member inventory. No
timestamp is embedded, so two builds are byte-identical.

CLI
---
``--mount-pack DIR``      unpacked frozen mount pack (contains MANIFEST.sha256)
``--release-id TEXT``     e.g. v0.1.1-preview.3 (informational stamp)
``--release-commit HEX40`` the commit the members must stamp
``--size-cap BYTES``      hard cap for the composed document (default 524288)
``--out FILE``            destination markdown file (must not exist)
"""
from __future__ import annotations

import argparse
import hashlib
import os
import re
import sys
from pathlib import Path

MANIFEST_NAME = "MANIFEST.sha256"
FALLBACK_SCHEMA = "PIPD-WEB-FALLBACK/1"
DEFAULT_SIZE_CAP = 524288
HEX40 = re.compile(r"^[0-9a-f]{40}$")


class Refusal(Exception):
    """A fail-closed refusal. ``code`` is the machine-checkable typed reason."""

    def __init__(self, code: str, detail: str) -> None:
        super().__init__(f"REFUSED {code}: {detail}")
        self.code = code
        self.detail = detail


def _path_is_symlink(path) -> bool:
    """Seam for tests: a single predicate the destination gate consults."""
    return os.path.islink(path)


def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def _canonical_bytes(data: bytes) -> bytes:
    if b"\x00" in data:
        return data
    try:
        data.decode("utf-8")
    except UnicodeDecodeError:
        return data
    return data.replace(b"\r\n", b"\n").replace(b"\r", b"\n")


def _is_text_member(data: bytes) -> bool:
    """A member is embeddable verbatim only if it is UTF-8 text with no NUL byte.

    The fallback document is markdown, so a non-text member could only be embedded through a lossy
    ``decode(..., "replace")`` that silently corrupts the source bytes. Fail closed instead.
    """
    if b"\x00" in data:
        return False
    try:
        data.decode("utf-8")
    except UnicodeDecodeError:
        return False
    return True


def _parse_manifest(text: str) -> list[tuple[str, str]]:
    entries: list[tuple[str, str]] = []
    for raw in text.splitlines():
        line = raw.rstrip("\n")
        if not line.strip():
            continue
        if "  " not in line:
            raise Refusal("MANIFEST_INVALID", f"line has no two-space separator: {line!r}")
        digest, rel = line.split("  ", 1)
        digest, rel = digest.strip(), rel.strip()
        if not re.fullmatch(r"[0-9a-f]{64}", digest):
            raise Refusal("MANIFEST_INVALID", f"digest is not 64 hex chars: {digest!r}")
        entries.append((digest, rel))
    if not entries:
        raise Refusal("MANIFEST_INVALID", "MANIFEST.sha256 lists no members")
    return entries


def _escapes(rel: str) -> bool:
    parts = rel.replace("\\", "/").split("/")
    return (rel.startswith("/") or "\\" in rel or ":" in rel or ".." in parts
            or any(part == "" for part in parts) or parts[0] in (".", ""))


def load_and_verify_pack(mount_pack: Path, release_commit: str,
                         release_id: str | None = None) -> list[tuple[str, str, bytes]]:
    """Return [(rel, sha256, bytes), ...] for a verified, exactly-listed pack."""
    mount_pack = Path(mount_pack)
    if _path_is_symlink(mount_pack):
        raise Refusal("MEMBER_SYMLINK", f"mount pack {mount_pack} is a symlink")
    if not mount_pack.is_dir():
        raise Refusal("MOUNT_PACK_MISSING", f"mount pack is not a directory: {mount_pack}")
    manifest_path = mount_pack / MANIFEST_NAME
    if _path_is_symlink(manifest_path):
        raise Refusal("MEMBER_SYMLINK", f"{manifest_path} is a symlink")
    if not manifest_path.is_file():
        raise Refusal("MANIFEST_MISSING", f"{manifest_path} is absent")
    entries = _parse_manifest(manifest_path.read_text(encoding="utf-8"))

    listed: list[tuple[str, str, bytes]] = []
    listed_paths: set[str] = set()
    for digest, rel in entries:
        if _escapes(rel):
            raise Refusal("PATH_ESCAPE", f"manifest path escapes the pack root: {rel!r}")
        member = mount_pack / rel
        if _path_is_symlink(member):
            raise Refusal("MEMBER_SYMLINK", f"{member} is a symlink")
        if not member.is_file():
            raise Refusal("MANIFEST_MEMBER_MISSING", f"manifest lists {rel} but it is absent")
        data = member.read_bytes()
        actual = _sha256(_canonical_bytes(data))
        if actual != digest:
            raise Refusal("MANIFEST_DIGEST_MISMATCH",
                          f"{rel}: copied digest {digest} != computed {actual}")
        if not _is_text_member(data):
            raise Refusal("MEMBER_NOT_TEXT",
                          f"{rel} is not UTF-8 text; refusing to embed it in a markdown document")
        listed_paths.add(rel)
        listed.append((rel, digest, data))

    on_disk: set[str] = set()
    for dirpath, dirnames, filenames in os.walk(mount_pack, followlinks=False):
        dirnames.sort()
        filenames.sort()
        for d in dirnames:
            if _path_is_symlink(Path(dirpath) / d):
                raise Refusal("MEMBER_SYMLINK", str(Path(dirpath) / d))
        for f in filenames:
            p = Path(dirpath) / f
            if _path_is_symlink(p):
                raise Refusal("MEMBER_SYMLINK", str(p))
            rel = p.relative_to(mount_pack).as_posix()
            if rel == MANIFEST_NAME:
                continue
            on_disk.add(rel)
    extra = sorted(on_disk - listed_paths)
    if extra:
        raise Refusal("UNEXPECTED_EXTRA_FILE", f"pack carries unlisted file(s): {extra}")

    for rel, _digest, data in listed:
        if not rel.endswith(".md"):
            continue
        text = data.decode("utf-8")
        if release_commit not in text:
            raise Refusal("RELEASE_STAMP_MISSING",
                          f"{rel} does not stamp the pinned release commit {release_commit}")
        if release_id and release_id not in text:
            raise Refusal("RELEASE_STAMP_MISSING",
                          f"{rel} does not stamp the release id {release_id}; a member that names "
                          "the pinned commit but not the release is a stale release stamp")
    return listed


def compose_document(release_id: str, release_commit: str,
                     members: list[tuple[str, str, bytes]], size_cap: int) -> bytes:
    """Compose the single bounded markdown document (no truncation anywhere)."""
    lines: list[str] = [
        f"<!-- {FALLBACK_SCHEMA} DOC_MOUNT_ONLY -->",
        "# PIPD-LS-SP S4 Web fallback — single-document mount (DOC_MOUNT_ONLY)",
        "",
        f"- release_id: {release_id}",
        f"- release_commit: {release_commit}",
        "- doc_mount_only: true",
        f"- member_count: {len(members)}",
        f"- size_cap_bytes: {size_cap}",
        "",
        "> DOC_MOUNT_ONLY: this is a document drop, not an installed plugin, not a host-registered",
        "> skill and not a public directory listing (none exists yet). It executes nothing. Verify a",
        "> member against the inventory below before you rely on it.",
        "",
        "## Member inventory",
        "",
        "| member | sha256 |",
        "| --- | --- |",
    ]
    for rel, digest, _data in members:
        lines.append(f"| {rel} | {digest} |")
    for rel, _digest, data in members:
        lines += ["", f"## Member: {rel}", "", f"<!-- BEGIN MEMBER {rel} -->"]
        lines.append(data.decode("utf-8", "replace"))
        lines.append(f"<!-- END MEMBER {rel} -->")
    text = "\n".join(lines)
    if not text.endswith("\n"):
        text += "\n"
    return text.encode("utf-8")


def write_exclusive(out_path: Path, data: bytes) -> None:
    out_path = Path(out_path)
    if _path_is_symlink(out_path):
        raise Refusal("DESTINATION_SYMLINK", f"destination {out_path} is a symlink")
    out_path.parent.mkdir(parents=True, exist_ok=True)
    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
    try:
        fd = os.open(out_path, flags, 0o644)
    except FileExistsError as exc:
        raise Refusal("DESTINATION_EXISTS",
                      f"destination {out_path} already exists; refusing to overwrite") from exc
    try:
        with os.fdopen(fd, "wb") as handle:
            handle.write(data)
    except BaseException:
        if out_path.exists() and not _path_is_symlink(out_path):
            out_path.unlink()
        raise


def build_fallback(*, mount_pack: Path, release_id: str, release_commit: str, out_path: Path,
                   size_cap: int = DEFAULT_SIZE_CAP) -> dict:
    if not HEX40.match(str(release_commit)):
        raise Refusal("RELEASE_COMMIT_INVALID",
                      f"release commit is not 40 hex chars: {release_commit!r}")
    if size_cap <= 0:
        raise Refusal("SIZE_CAP_INVALID", f"size cap must be positive: {size_cap!r}")
    members = load_and_verify_pack(mount_pack, release_commit, release_id)
    document = compose_document(release_id, release_commit, members, size_cap)
    if len(document) > size_cap:
        raise Refusal("SIZE_CAP_EXCEEDED",
                      f"composed document is {len(document)} bytes > cap {size_cap}; "
                      "refusing to truncate")
    write_exclusive(Path(out_path), document)
    return {
        "verdict": "BUILT",
        "out": str(out_path),
        "bytes": len(document),
        "sha256": _sha256(document),
        "member_count": len(members),
        "doc_mount_only": True,
    }

#!/usr/bin/env python3
"""S4POST Stage B: freeze the eight public skill packs into a source ledger.

The public plugin builder (``build_public_plugin.py``) may only consume the eight ``skills/pipd-*``
packs through a frozen ledger produced here from a *clean checkout*. The ledger pins the repo, the
commit (40 hex), the tree (40 hex), the exact eight skill names and a ``path -> sha256`` for every
file under them, and it is the binding the owner derivation records must match
(``source_sha256``). This tool REFUSES (typed reason, non-zero exit) instead of warning:

  * ``DIRTY_WORKTREE``          git status --porcelain is not empty
  * ``NOT_REPO_ROOT``           the given root is not the git top-level
  * ``GIT_UNAVAILABLE`` / ``GIT_FAILED``  git could not run
  * ``WRONG_SKILL_SET``         the skills root is missing or carries an unexpected skill
  * ``SYMLINK_MEMBER``          a symlinked file or directory inside a skill
  * ``NON_REGULAR_FILE``        a special/irregular file inside a skill (FIFO, socket, ...)
  * ``HIDDEN_OR_SPECIAL_MEMBER`` a dot-file/dot-directory inside a skill
  * ``SKILLS_ROOT_MISSING``     the skills root does not exist

Text members are canonicalised to LF (the repository pins ``* text=auto eol=lf``) so the same
checkout hashes identically on any platform; binary members are hashed raw. No git history is
written, no commit is created, nothing is published.

CLI
---
``--repo-root DIR``            repository root (default: the directory of this file)
``--skills-root DIR``          override the skills root (default: <repo>/skills)
``--out FILE``                 write the ledger JSON to FILE (default: stdout only)
``--no-git``                   skip the git clean/commit/tree gates (fixtures; requires the
                               ``--ledger-commit`` and ``--ledger-tree`` overrides)
``--ledger-commit HEX40``      freeze this commit instead of the git HEAD (fixtures/audits)
``--ledger-tree HEX40``        freeze this tree instead of the git HEAD^{tree}
"""
from __future__ import annotations

import hashlib
import json
import os

import shutil

import sys

LEDGER_VERSION = "PIPD-PUBLIC-PLUGIN-SOURCE-LEDGER/1"
EXPECTED_SKILLS = (
    "pipd-assurance-tqaep",
    "pipd-authority-source",
    "pipd-execution-contract",

class FreezeRefusal(Exception):
    """A fail-closed refusal. ``code`` is the machine-checkable typed reason."""

    def __init__(self, code: str, detail: str) -> None:
        super().__init__(f"REFUSED {code}: {detail}")
        self.code = code
        self.detail = detail


def _sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def canonical_bytes(data: bytes) -> bytes:
    """Canonical member bytes: LF-normalised for text, raw for binary."""
    if b"\x00" in data:
        return data
    try:
        data.decode("utf-8")
    except UnicodeDecodeError:
        return data
    return data.replace(b"\r\n", b"\n").replace(b"\r", b"\n")


def _git(repo: Path, *args: str) -> str:
    git = shutil.which("git")
    if git is None:
        raise FreezeRefusal("GIT_UNAVAILABLE", "no `git` executable on PATH")
    proc = subprocess.run([git, "-C", str(repo), *args], capture_output=True, text=True)
    if proc.returncode != 0:
        raise FreezeRefusal(
            "GIT_FAILED",
            f"`git {' '.join(args)}` exited {proc.returncode}: "
            f"{(proc.stderr or proc.stdout).strip()}",
        )
    return proc.stdout


def _walk_skill_files(repo_root: Path, skill_dir: Path) -> dict[str, str]:
    """Every regular file under one skill, as ``repo-relative POSIX path -> canonical sha256``."""
    files: dict[str, str] = {}
    for dirpath, dirnames, filenames in os.walk(skill_dir, topdown=True, followlinks=False):
        dirnames.sort()
        filenames.sort()
        for d in list(dirnames):
            p = Path(dirpath) / d
            if p.is_symlink():
                raise FreezeRefusal("SYMLINK_MEMBER", str(p))
            if d.startswith(".") or d in ("__pycache__",):
                raise FreezeRefusal("HIDDEN_OR_SPECIAL_MEMBER", str(p))
        for f in filenames:
            p = Path(dirpath) / f
            st = os.lstat(p)
            if stat.S_ISLNK(st.st_mode):
                raise FreezeRefusal("SYMLINK_MEMBER", str(p))
            if not stat.S_ISREG(st.st_mode):
                raise FreezeRefusal(
                    "NON_REGULAR_FILE",
                    f"{p} is not a regular file (mode {oct(stat.S_IMODE(st.st_mode))})")
            if f.startswith(".") or f in ("__pycache__",):
                raise FreezeRefusal("HIDDEN_OR_SPECIAL_MEMBER", str(p))
            rel = p.relative_to(repo_root).as_posix()
            files[rel] = _sha256_bytes(canonical_bytes(p.read_bytes()))
    return files


def _skill_entries(skills_root: Path) -> dict[str, Path]:
    if not skills_root.is_dir():
        raise FreezeRefusal("SKILLS_ROOT_MISSING", str(skills_root))
    entries: dict[str, Path] = {}
    for child in sorted(skills_root.iterdir()):
        if child.name.startswith(".") or child.name == "__pycache__":
            continue
        if child.is_symlink():
            raise FreezeRefusal("SYMLINK_MEMBER", str(child))
        if not child.is_dir():
            raise FreezeRefusal("NON_REGULAR_FILE",
                                f"{child} sits directly in the skills root and is not a skill dir")
        entries[child.name] = child
    return entries


def freeze(repo_root: Path, skills_root: Path | None = None, *, require_git: bool = True,
           commit_override: str | None = None, tree_override: str | None = None,
           repo_url_override: str | None = None,
           expected_skills: tuple[str, ...] = EXPECTED_SKILLS) -> dict:
    """Produce the frozen source ledger. Raises :class:`FreezeRefusal` on any deviation."""
    repo_root = Path(repo_root).resolve()
    skills_root = Path(skills_root).resolve() if skills_root is not None else repo_root / "skills"

    if require_git:
        top = _git(repo_root, "rev-parse", "--show-toplevel").strip()
        if Path(top).resolve() != repo_root:
            raise FreezeRefusal("NOT_REPO_ROOT",
                                f"git top-level {top!r} != given repo root {repo_root}")
        dirty = _git(repo_root, "status", "--porcelain", "--untracked-files=normal").strip()
        if dirty:
            raise FreezeRefusal("DIRTY_WORKTREE",
                                "worktree is not clean:\n" + "\n".join(dirty.splitlines()[:20]))
        commit = _git(repo_root, "rev-parse", "HEAD").strip()
        tree = _git(repo_root, "rev-parse", "HEAD^{tree}").strip()
        repo_url = _git(repo_root, "remote", "get-url", "origin").strip() or "NO_REMOTE"
    else:
        if commit_override is None or tree_override is None:
            raise FreezeRefusal("FIXTURE_OVERRIDE_MISSING",
                                "--no-git requires --ledger-commit and --ledger-tree")
        commit, tree = commit_override, tree_override
        repo_url = repo_url_override or "FIXTURE_NO_GIT"

    for label, value in (("commit", commit), ("tree", tree)):
        if not HEX40.match(value):
            raise FreezeRefusal("IDENTITY_INVALID",
                                f"{label} is not 40 hex chars: {value!r}")

    entries = _skill_entries(skills_root)
    missing = [n for n in expected_skills if n not in entries]
    extra = [n for n in sorted(entries) if n not in set(expected_skills)]
    if missing or extra:
        raise FreezeRefusal("WRONG_SKILL_SET",
                            f"expected exactly {sorted(expected_skills)}; "
                            f"missing={missing} extra={extra}")

    files: dict[str, str] = {}
    for name in sorted(entries):
        files.update(_walk_skill_files(repo_root, entries[name]))

    total = sum(len(canonical_bytes((repo_root / rel).read_bytes())) for rel in files)
    return {
        "ledger_version": LEDGER_VERSION,
        "repo": repo_url,
        "commit": commit,
        "tree": tree,
        "skills": sorted(entries),
        "canonicalization": "text=LF;binary=raw",
        "file_count": len(files),
        "total_canonical_bytes": total,
        "files": files,
    }


def canonical_json(doc: dict) -> str:
    return json.dumps(doc, ensure_ascii=False, indent=2, sort_keys=True) + "\n"


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    ap.add_argument("--repo-root", default=str(Path(__file__).resolve().parents[1]))
    ap.add_argument("--skills-root", default=None)
    ap.add_argument("--out", default=None)
    ap.add_argument("--no-git", action="store_true", help="fixture mode (no git gates)")
    ap.add_argument("--ledger-commit", default=None)
    ap.add_argument("--ledger-tree", default=None)
    ap.add_argument("--repo-url", default=None)
    args = ap.parse_args(argv)
    try:
        ledger = freeze(
            Path(args.repo_root),
            Path(args.skills_root) if args.skills_root else None,
            require_git=not args.no_git,
            commit_override=args.ledger_commit,
            tree_override=args.ledger_tree,
            repo_url_override=args.repo_url,
        )
    except FreezeRefusal as exc:
        print(str(exc))
        return 2
    text = canonical_json(ledger)
    if args.out:
        out = Path(args.out)
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_text(text, encoding="utf-8", newline="")
    sys.stdout.write(text)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

@contextlib.contextmanager
def as_symlink(path: Path, target: Path):
    """Present ``path`` as a symlink to ``target``; fall back to the documented seam.

    The seam (``build_public_plugin._path_is_symlink``) exists precisely so the symlink gates can
    be exercised on hosts that forbid real symlinks. Only the predicate is substituted; every
    other line of the gate is the real, unmodified code.
    """
    try:
        os.symlink(target, path, target_is_directory=target.is_dir())
    except (OSError, NotImplementedError):
        real = B._path_is_symlink
        wanted = Path(path)

        def fake(candidate, _wanted=wanted, _real=real):
            try:
                if Path(candidate) == _wanted:
                    return True
            except TypeError:
                pass
            return _real(candidate)

        B._path_is_symlink = fake
        try:
            yield
        finally:
            B._path_is_symlink = real
    else:
        try:
            yield
        finally:
            try:
                path.unlink()
            except OSError:
                pass


def zip_entries(path: Path) -> tuple[list[zipfile.ZipInfo], dict[str, bytes]]:
    with zipfile.ZipFile(path) as archive:
        infos = archive.infolist()
        members = {info.filename: archive.read(info.filename) for info in infos}
    return infos, members

"""Stage B lane B2: the Web fallback single-document projection.

Exercises ``tools/build_web_fallback.py`` end to end against temp-directory mount packs. Nothing
here touches the network or the real repository tree: every fixture is authored under a
``tempfile.TemporaryDirectory`` and every pack is a small ``MANIFEST.sha256`` plus its members.

The tool reproduces one frozen mount pack into ONE bounded markdown document
(``DOC_MOUNT_ONLY``) and is fail-closed. The cases below pin the positive exact projection and the
refusals:

 1. exact projection: verbatim member bytes, member inventory table, ``DOC_MOUNT_ONLY`` stamp
 2. stale release stamp (only the release id, or only the pinned commit, is REFUSED)
 3. over-cap composition REFUSED, and the destination is never created (no truncation)
 4. member digest disagreeing with ``MANIFEST.sha256`` REFUSED
 5. an unexpected extra file REFUSED; a manifest entry with no file REFUSED
 6. an existing destination REFUSED with its bytes unchanged (``O_EXCL``)
 7. a symlinked destination REFUSED, with the file behind it byte-identical
 8. a symlinked manifest / symlinked member REFUSED; a member path escaping the root REFUSED
 9. a non-UTF-8 / non-text member REFUSED
10. the CLI ``main()`` refusal path exits non-zero and refuses above-cap output instead of trimming

The fixtures are authored as LF-only UTF-8, which is exactly the canonical byte sequence the pack
builder ``tools/build_mount_pack.py`` hashes (EOL-normalised text members), so a member's manifest
digest is simply ``sha256`` of the bytes written here.
"""

# --------------------------------------------------------------------------- fixture helpers

def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def _manifest_text(entries: list[tuple[str, bytes]]) -> str:
    return "".join(f"{_sha256(data)}  {rel}\n" for rel, data in entries)


def _write_pack(root: Path, members: dict[str, bytes], *, manifest: str | None = None) -> Path:
    """Author a mount pack: member files (LF bytes verbatim) plus a derived ``MANIFEST.sha256``."""
    root.mkdir(parents=True, exist_ok=True)
    for rel, data in members.items():
        member = root / rel
        member.parent.mkdir(parents=True, exist_ok=True)
        member.write_bytes(data)
    text = manifest if manifest is not None else _manifest_text(list(members.items()))
    (root / WFB.MANIFEST_NAME).write_bytes(text.encode("utf-8"))
    return root


def _stamp(name: str, extra: str = "") -> bytes:
    """A member document carrying BOTH the pinned release id and the pinned commit."""
    text = (f"# {name}\n\n"
            f"- release_id: {RELEASE_ID}\n"
            f"- release_commit: {RELEASE_COMMIT}\n\n"
            f"Verbatim body for {name}.\n{extra}")
    return text.encode("utf-8")


_SYMLINK_PROBE: list = [None]


def _symlinks_supported() -> bool:
    """Whether this host can actually create a symlink (Windows often cannot without privilege)."""
    if _SYMLINK_PROBE[0] is None:
        with tempfile.TemporaryDirectory() as tmp:
            target = Path(tmp) / "target"
            target.write_bytes(b"x")
            link = Path(tmp) / "link"
            try:
                os.symlink(target, link)
                _SYMLINK_PROBE[0] = os.path.islink(link)
            except (OSError, NotImplementedError, AttributeError):
                _SYMLINK_PROBE[0] = False
    return bool(_SYMLINK_PROBE[0])


@contextlib.contextmanager
def _seam_symlink(path: Path):
    """Report exactly ``path`` as a symlink through the tool's documented ``_path_is_symlink`` seam.

    Used only when the host cannot create a real symlink; the refusal gate itself is unchanged.
    """
    seam_path = os.path.abspath(str(path))
    real = WFB._path_is_symlink

    def fake(candidate):
        return os.path.abspath(str(candidate)) == seam_path or real(candidate)

    WFB._path_is_symlink = fake
    try:
        yield
    finally:
        WFB._path_is_symlink = real

This page is the **install SSOT**: every other page that mentions installing PIPD-LS-SP either is

Version pin for everything on this page: release **`v0.1.1-preview.3`**, release-tag target and

There is **no public plugin directory listing yet**: nothing has been published to a plugin

## The Web fallback document and its `DOC_MOUNT_ONLY` ceiling

`PIPD_WEB_FALLBACK.md` is one markdown document built from a *frozen mount pack* — a directory that

The tool is fail-closed, not best-effort. It **refuses** (typed reason, non-zero exit) instead of
